/* ============================================================ VIAVIA AIRLINES — AVAILABLE BIDS Uses the shared pairing engine from viavia-schedule.js IMPORTANT: - Do NOT generate pairings locally. - Do NOT use Math.random(). - viavia-schedule.js is the single source of truth. - All available trips come from generateViaviaPairings(). - Bid selections are saved in localStorage. ============================================================ */ const STORAGE_KEY = "viaviaBids"; let allPairings = []; let filteredPairings = []; /* ============================================================ INITIALIZATION ============================================================ */ document.addEventListener("DOMContentLoaded", () => { initializeAvailableBids(); }); function initializeAvailableBids() { if ( typeof VIAVIA_SCHEDULE === "undefined" || typeof generateViaviaPairings !== "function" ) { console.error( "Viavia schedule or pairing engine was not loaded." ); showScheduleError(); return; } loadPairings(); setupFilters(); applyFilters(); } /* ============================================================ LOAD SHARED PAIRINGS ============================================================ */ function loadPairings() { /* * The pairing engine lives in viavia-schedule.js. * * Available Bids does NOT create its own trips. */ allPairings = generateViaviaPairings({ maxLegs: 3 }); /* * Make a clean copy for filtering. */ filteredPairings = [...allPairings]; } /* ============================================================ FILTER SETUP ============================================================ */ function setupFilters() { const baseFilter = document.getElementById( "baseFilter" ); const fleetFilter = document.getElementById( "fleetFilter" ); const lengthFilter = document.getElementById( "lengthFilter" ); const searchInput = document.getElementById( "searchInput" ); if (baseFilter) { baseFilter.addEventListener( "change", applyFilters ); } if (fleetFilter) { fleetFilter.addEventListener( "change", applyFilters ); } if (lengthFilter) { lengthFilter.addEventListener( "change", applyFilters ); } if (searchInput) { searchInput.addEventListener( "input", applyFilters ); } } /* ============================================================ APPLY FILTERS ============================================================ */ function applyFilters() { const baseFilter = document.getElementById( "baseFilter" )?.value || "ALL"; const fleetFilter = document.getElementById( "fleetFilter" )?.value || "ALL"; const lengthFilter = document.getElementById( "lengthFilter" )?.value || "ALL"; const searchInput = document.getElementById( "searchInput" )?.value .trim() .toUpperCase() || ""; filteredPairings = allPairings.filter( pairing => { /* -------------------------------------------- BASE FILTER -------------------------------------------- */ if ( baseFilter !== "ALL" && pairing.base !== baseFilter ) { return false; } /* -------------------------------------------- FLEET FILTER -------------------------------------------- */ if ( fleetFilter !== "ALL" ) { if ( !pairing.aircraft.includes( fleetFilter ) ) { return false; } } /* -------------------------------------------- TRIP LENGTH FILTER -------------------------------------------- */ if ( lengthFilter !== "ALL" ) { const requiredLength = Number( lengthFilter ); if ( pairing.legs !== requiredLength ) { return false; } } /* -------------------------------------------- SEARCH FILTER -------------------------------------------- */ if ( searchInput ) { const searchableText = [ pairing.pairingId, pairing.base, pairing.route, ...pairing.flightNumbers, ...pairing.flights.map( flight => flight.flightNumber ), ...pairing.flights.map( flight => flight.origin ), ...pairing.flights.map( flight => flight.destination ), ...pairing.flights.map( flight => flight.aircraft ) ] .join(" ") .toUpperCase(); if ( !searchableText.includes( searchInput ) ) { return false; } } return true; } ); renderPairings(); } /* ============================================================ RENDER PAIRINGS ============================================================ */ function renderPairings() { const container = document.getElementById( "bidsContainer" ); if (!container) { console.warn( "bidsContainer was not found." ); return; } if ( filteredPairings.length === 0 ) { container.innerHTML = `
✈

No Available Trips

No pairings match your current filters.

`; updateBidCount(); return; } container.innerHTML = filteredPairings .map( pairing => createPairingCard( pairing ) ) .join(""); updateBidCount(); } /* ============================================================ CREATE PAIRING CARD ============================================================ */ function createPairingCard( pairing ) { const firstFlight = pairing.flights[0]; const lastFlight = pairing.flights[ pairing.flights.length - 1 ]; const aircraft = pairing.aircraft.join( " / " ); const flightRows = pairing.flights .map( flight => createFlightRow( flight ) ) .join(""); const savedBids = getSavedBids(); const isBidSubmitted = savedBids.some( bid => bid.pairingId === pairing.pairingId ); return `
${escapeHtml( pairing.pairingId )}

${escapeHtml( pairing.route )}

${escapeHtml( pairing.base )}
Legs ${pairing.legs}
Aircraft ${escapeHtml( aircraft )}
Status ${ pairing.closed ? "Closed Turn" : "Open Trip" }
${flightRows}
Gate Assignment ${escapeHtml( VIAVIA_SCHEDULE .operations .gatePendingText )}
Starts: ${escapeHtml( firstFlight.origin )} Ends: ${escapeHtml( lastFlight.destination )}
`; } /* ============================================================ CREATE FLIGHT ROW ============================================================ */ function createFlightRow( flight ) { return `
${escapeHtml( flight.flightNumber )} ${escapeHtml( flight.aircraft )}
${escapeHtml( flight.origin )} ${escapeHtml( flight.departure )}
✈
${escapeHtml( flight.destination )} ${escapeHtml( flight.arrival )} ${ flight.nextDay ? "+1" : "" }
`; } /* ============================================================ BID STORAGE ============================================================ */ function getSavedBids() { try { const stored = localStorage.getItem( STORAGE_KEY ); if (!stored) { return []; } const parsed = JSON.parse( stored ); if ( !Array.isArray( parsed ) ) { return []; } return parsed; } catch (error) { console.error( "Unable to read Viavia bids:", error ); return []; } } /* ============================================================ SAVE BIDS ============================================================ */ function saveBids( bids ) { try { localStorage.setItem( STORAGE_KEY, JSON.stringify( bids ) ); return true; } catch (error) { console.error( "Unable to save Viavia bids:", error ); return false; } } /* ============================================================ SUBMIT BID ============================================================ */ function submitBid( pairingId ) { const pairing = allPairings.find( item => item.pairingId === pairingId ); if (!pairing) { alert( "Unable to find this trip." ); return; } const savedBids = getSavedBids(); const alreadySubmitted = savedBids.some( bid => bid.pairingId === pairingId ); if ( alreadySubmitted ) { return; } const bidRecord = { pairingId: pairing.pairingId, base: pairing.base, legs: pairing.legs, flightNumbers: [ ...pairing.flightNumbers ], route: pairing.route, aircraft: [ ...pairing.aircraft ], submittedAt: new Date().toISOString(), status: "SUBMITTED" }; savedBids.push( bidRecord ); const saved = saveBids( savedBids ); if (!saved) { alert( "Your bid could not be saved. Please try again." ); return; } renderPairings(); showBidConfirmation( pairing ); } /* ============================================================ BID CONFIRMATION ============================================================ */ function showBidConfirmation( pairing ) { alert( "Bid submitted successfully.\n\n" + `${pairing.pairingId}\n` + `${pairing.route}\n\n` + `${pairing.flightNumbers.join( " → " )}` ); } /* ============================================================ BID COUNT ============================================================ */ function updateBidCount() { const countElement = document.getElementById( "bidCount" ); if (!countElement) { return; } const count = filteredPairings.length; countElement.textContent = `${count} ${ count === 1 ? "trip" : "trips" } available`; } /* ============================================================ REFRESH AVAILABLE BIDS ============================================================ */ function refreshAvailableBids() { loadPairings(); applyFilters(); } /* ============================================================ SCHEDULE ERROR ============================================================ */ function showScheduleError() { const container = document.getElementById( "bidsContainer" ); if (!container) { return; } container.innerHTML = `
⚠

Schedule Unavailable

The Viavia master schedule could not be loaded.

Please refresh the page and try again.

`; } /* ============================================================ HTML ESCAPING ============================================================ */ function escapeHtml( value ) { if ( value === null || value === undefined ) { return ""; } return String(value) .replace( /&/g, "&" ) .replace( //g, ">" ) .replace( /"/g, """ ) .replace( /'/g, "'" ); } /* ============================================================ DEBUG FUNCTIONS ============================================================ */ /* * These remain available from the browser console. * * allPairings * * filteredPairings * * getSavedBids() * * refreshAvailableBids() * * generateViaviaPairings() */ /* ============================================================ FINAL SAFETY CHECK ============================================================ */ if ( typeof generateViaviaPairings !== "function" ) { console.error( "Viavia pairing engine was not loaded. " + "Make sure viavia-schedule.js is loaded " + "before the Available Bids script." ); }

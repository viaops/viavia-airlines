/* ============================================================ VIAVIA AVAILABLE BIDS Uses the shared pairing engine from viavia-schedule.js ============================================================ */ const STORAGE_KEY = "viaviaBids"; let allPairings = []; let filteredPairings = []; /* ============================================================ INITIALIZATION ============================================================ */ document.addEventListener("DOMContentLoaded", () => { loadPairings(); setupFilters(); renderPairings(); }); /* ============================================================ LOAD SHARED PAIRINGS ============================================================ */ function loadPairings() { /* * generateViaviaPairings() comes directly from * viavia-schedule.js. * * There is intentionally NO local pairing generator here. */ allPairings = generateViaviaPairings({ maxLegs: 3 }); filteredPairings = [...allPairings]; } /* ============================================================ FILTER SETUP ============================================================ */ function setupFilters() { const baseFilter = document.getElementById("baseFilter"); const fleetFilter = document.getElementById("fleetFilter"); const lengthFilter = document.getElementById("lengthFilter"); const searchInput = document.getElementById("searchInput"); if (baseFilter) { baseFilter.addEventListener( "change", applyFilters ); } if (fleetFilter) { fleetFilter.addEventListener( "change", applyFilters ); } if (lengthFilter) { lengthFilter.addEventListener( "change", applyFilters ); } if (searchInput) { searchInput.addEventListener( "input", applyFilters ); } } /* ============================================================ FILTER PAIRINGS ============================================================ */ function applyFilters() { const baseFilter = document.getElementById( "baseFilter" )?.value || "ALL"; const fleetFilter = document.getElementById( "fleetFilter" )?.value || "ALL"; const lengthFilter = document.getElementById( "lengthFilter" )?.value || "ALL"; const searchInput = document.getElementById( "searchInput" )?.value .trim() .toUpperCase() || ""; filteredPairings = allPairings.filter( pairing => { /* -------------------------------------------- BASE -------------------------------------------- */ if ( baseFilter !== "ALL" && pairing.base !== baseFilter ) { return false; } /* -------------------------------------------- FLEET -------------------------------------------- */ if ( fleetFilter !== "ALL" ) { const fleetMatch = pairing.aircraft.includes( fleetFilter ); if (!fleetMatch) { return false; } } /* -------------------------------------------- TRIP LENGTH -------------------------------------------- */ if ( lengthFilter !== "ALL" ) { const requiredLength = Number( lengthFilter ); if ( pairing.legs !== requiredLength ) { return false; } } /* -------------------------------------------- SEARCH -------------------------------------------- */ if (searchInput) { const searchableText = [ pairing.pairingId, pairing.base, pairing.route, ...pairing.flightNumbers, ...pairing.flights.map( flight => flight.origin ), ...pairing.flights.map( flight => flight.destination ) ] .join(" ") .toUpperCase(); if ( !searchableText.includes( searchInput ) ) { return false; } } return true; } ); renderPairings(); } /* ============================================================ RENDER PAIRINGS ============================================================ */ function renderPairings() { const container = document.getElementById( "bidsContainer" ); if (!container) { return; } if ( filteredPairings.length === 0 ) { container.innerHTML = `
✈

No Available Trips

No pairings match your current filters.

`; updateBidCount(); return; } container.innerHTML = filteredPairings .map( pairing => createPairingCard( pairing ) ) .join(""); updateBidCount(); } /* ============================================================ CREATE PAIRING CARD ============================================================ */ function createPairingCard( pairing ) { const firstFlight = pairing.flights[0]; const lastFlight = pairing.flights[ pairing.flights.length - 1 ]; const aircraft = pairing.aircraft.join( " / " ); const flightRows = pairing.flights .map( ( flight, index ) => createFlightRow( flight, index ) ) .join(""); const savedBids = getSavedBids(); const isBidSubmitted = savedBids.some( bid => bid.pairingId === pairing.pairingId ); return `
${pairing.pairingId}

${pairing.route}

${pairing.base}
Legs ${pairing.legs}
Aircraft ${aircraft}
Status ${pairing.closed ? "Closed Turn" : "Open Trip"}
${flightRows}
Gate Assignment ${VIAVIA_SCHEDULE .operations .gatePendingText}
Starts: ${firstFlight.origin} Ends: ${lastFlight.destination}
`; } /* ============================================================ CREATE FLIGHT ROW ============================================================ */ function createFlightRow( flight, index ) { const connection = index > 0 ? getViaviaConnectionMinutes( pairingFlightPlaceholder( flight ), flight ) : null; return `
${flight.flightNumber} ${flight.aircraft}
${flight.origin} ${flight.departure}
✈
${flight.destination} ${flight.arrival} ${ flight.nextDay ? "+1" : "" }
`; } /* * This is only used to prevent the connection helper * from being called incorrectly while rendering. * * The actual pairing connection is already validated * by viavia-schedule.js. */ function pairingFlightPlaceholder( flight ) { return flight; } /* ============================================================ BID STORAGE ============================================================ */ function getSavedBids() { try { const stored = localStorage.getItem( STORAGE_KEY ); if (!stored) { return []; } const parsed = JSON.parse( stored ); return Array.isArray(parsed) ? parsed : []; } catch (error) { console.error( "Unable to read Viavia bids:", error ); return []; } } /* ============================================================ SUBMIT BID ============================================================ */ function submitBid( pairingId ) { const pairing = allPairings.find( item => item.pairingId === pairingId ); if (!pairing) { alert( "Unable to find this trip." ); return; } const savedBids = getSavedBids(); const alreadySubmitted = savedBids.some( bid => bid.pairingId === pairingId ); if (alreadySubmitted) { return; } const bidRecord = { pairingId: pairing.pairingId, base: pairing.base, legs: pairing.legs, flightNumbers: [...pairing.flightNumbers], route: pairing.route, aircraft: [...pairing.aircraft], submittedAt: new Date().toISOString(), status: "SUBMITTED" }; savedBids.push( bidRecord ); localStorage.setItem( STORAGE_KEY, JSON.stringify( savedBids ) ); renderPairings(); showBidConfirmation( pairing ); } /* ============================================================ BID CONFIRMATION ============================================================ */ function showBidConfirmation( pairing ) { const message = ` Bid submitted successfully. ${pairing.pairingId} ${pairing.route} ${pairing.flightNumbers.join( " → " )} `; alert( message ); } /* ============================================================ BID COUNT ============================================================ */ function updateBidCount() { const countElement = document.getElementById( "bidCount" ); if (!countElement) { return; } countElement.textContent = `${filteredPairings.length} ${ filteredPairings.length === 1 ? "trip" : "trips" } available`; } /* ============================================================ REFRESH ============================================================ */ function refreshAvailableBids() { loadPairings(); applyFilters(); } /* ============================================================ DEBUG / DEVELOPMENT ============================================================ */ /* * Available from the browser console: * * allPairings * * filteredPairings * * getSavedBids() * * refreshAvailableBids() * */ /* ============================================================ SAFETY CHECK ============================================================ */ if ( typeof generateViaviaPairings !== "function" ) { console.error( "Viavia pairing engine was not loaded. " + "Make sure viavia-schedule.js appears " + "before this script." ); }
